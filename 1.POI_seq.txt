# -*- coding: utf-8 -*-
"""
Construct POI sequences around road-network sample points.

Input files:
  data/road_network/gis_osm_roads_free_1.shp
  data/poi_shp/*.shp

Output files:
  outputs/road_random_points.shp
  outputs/random_point_sjoin_poi/*_random_poi_i_use.pkl
  outputs/random_id_mid_type_seq3.pkl
  outputs/random_id_mid_type_seq6.pkl
  outputs/random_id_mid_type_seq8.pkl
"""

from pathlib import Path
import math
import pickle
import warnings

import geopandas as gpd
import numpy as np
import pandas as pd
from shapely.geometry import LineString, MultiLineString, Point

warnings.filterwarnings("ignore")

BASE_DIR = Path(".")
DATA_DIR = BASE_DIR / "data"
OUTPUT_DIR = BASE_DIR / "outputs"
PROJECT_CRS = "EPSG:4527"

ROAD_PATH = DATA_DIR / "road_network" / "gis_osm_roads_free_1.shp"
POI_DIR = DATA_DIR / "poi_shp"
RANDOM_POINTS_PATH = OUTPUT_DIR / "road_random_points.shp"
JOIN_OUTPUT_DIR = OUTPUT_DIR / "random_point_sjoin_poi"


def iter_lines(geometry):
    if geometry is None or geometry.is_empty:
        return
    if isinstance(geometry, LineString):
        yield geometry
    elif isinstance(geometry, MultiLineString):
        yield from geometry.geoms


def create_network_sample_points(road_path: Path, interval_m: int = 50) -> gpd.GeoDataFrame:
    road = gpd.read_file(road_path).to_crs(PROJECT_CRS)
    points = []
    for geom in road.geometry:
        for line in iter_lines(geom):
            distances = np.arange(0, line.length, interval_m)
            points.extend(line.interpolate(distance) for distance in distances)

    random_points = gpd.GeoDataFrame(geometry=points, crs=PROJECT_CRS)
    random_points["x_val"] = random_points.geometry.apply(lambda p: p.coords[0][0])
    random_points["y_val"] = random_points.geometry.apply(lambda p: p.coords[0][1])

    xmin = random_points["x_val"].min()
    ymin = random_points["y_val"].min()
    for shift_m in (0, interval_m / 2):
        random_points["x_grid"] = ((random_points["x_val"] - xmin + shift_m) / interval_m).apply(math.floor)
        random_points["y_grid"] = ((random_points["y_val"] - ymin + shift_m) / interval_m).apply(math.floor)
        random_points = random_points.drop_duplicates(subset=["x_grid", "y_grid"])

    random_points["geometry"] = random_points.apply(lambda row: Point(row.x_val, row.y_val), axis=1)
    random_points = random_points.reset_index(drop=True)
    random_points["ID"] = range(len(random_points))
    return random_points[["ID", "geometry"]]


def first_existing_column(frame: pd.DataFrame, candidates: list[str]) -> str:
    for col in candidates:
        if col in frame.columns:
            return col
    raise KeyError(f"None of the candidate columns exists: {candidates}")


def normalize_poi_columns(poi: gpd.GeoDataFrame) -> gpd.GeoDataFrame:
    mid_col = first_existing_column(poi, ["mid_class", "class_mid", "class", "中"])
    sub_col = next((col for col in ["sub_class", "小"] if col in poi.columns), mid_col)
    out = poi[[mid_col, sub_col, "geometry"]].copy()
    out = out.rename(columns={mid_col: "mid_class", sub_col: "sub_class"})
    out = out[out["mid_class"].apply(lambda x: isinstance(x, str))]
    return out


def join_poi_to_sample_points(random_points: gpd.GeoDataFrame, poi_dir: Path, buffer_m: int = 200) -> None:
    JOIN_OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    point_buffer = gpd.GeoDataFrame(
        data=random_points[["ID"]],
        geometry=random_points.buffer(buffer_m),
        crs=PROJECT_CRS,
    )
    point_geometry = random_points[["ID", "geometry"]].rename(columns={"geometry": "sample_geometry"})

    for poi_path in sorted(poi_dir.glob("*.shp")):
        poi = normalize_poi_columns(gpd.read_file(poi_path).to_crs(PROJECT_CRS))
        poi["poi_id"] = range(len(poi))
        joined = gpd.sjoin(point_buffer, poi, how="inner", predicate="intersects")
        poi_geometry = poi[["poi_id", "geometry"]].rename(columns={"geometry": "poi_geometry"})
        joined = joined.merge(point_geometry, on="ID", how="left").merge(poi_geometry, on="poi_id", how="left")
        joined["dist"] = joined.sample_geometry.distance(joined.poi_geometry)
        out_path = JOIN_OUTPUT_DIR / f"{poi_path.stem}_random_poi_i_use.pkl"
        with out_path.open("wb") as f:
            pickle.dump(joined[["ID", "mid_class", "sub_class", "dist"]], f)


def build_sequences(join_dir: Path) -> pd.DataFrame:
    records = []
    for pkl_path in sorted(join_dir.glob("*_random_poi_i_use.pkl")):
        with pkl_path.open("rb") as f:
            records.append(pickle.load(f)[["ID", "mid_class", "sub_class", "dist"]])
    random_poi = pd.concat(records, ignore_index=True)

    def poi_sequence(group: pd.DataFrame) -> list[str]:
        return group.sort_values("dist")["mid_class"].tolist()

    seq = random_poi.groupby("ID").apply(poi_sequence).reset_index()
    seq.columns = ["ID", "type_seq"]
    seq["quantity"] = seq["type_seq"].apply(len)
    return seq


def main() -> None:
    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    random_points = create_network_sample_points(ROAD_PATH, interval_m=50)
    random_points.to_file(RANDOM_POINTS_PATH)
    join_poi_to_sample_points(random_points, POI_DIR, buffer_m=200)

    sequences = build_sequences(JOIN_OUTPUT_DIR)
    for threshold in (3, 6, 8):
        out_path = OUTPUT_DIR / f"random_id_mid_type_seq{threshold}.pkl"
        with out_path.open("wb") as f:
            pickle.dump(sequences[sequences["quantity"] > threshold], f)


if __name__ == "__main__":
    main()
