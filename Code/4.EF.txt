# -*- coding: utf-8 -*-
"""
Compute enrichment factors (EF) for POI categories within Doc2Vec-K-means functional zones.

Input files:
  data/community_units/beijing_community.shp
  data/poi_shp/*.shp
  outputs/community_doc2vec_kmeans.shp

Output files:
  outputs/poi_all.pkl
  outputs/city_area_poi_english.pkl
  outputs/cluster_poi_type_seq3_mid_gat2_class_english.pkl
  outputs/EF_seq3_gat_2_class_english.xlsx
"""

from ast import literal_eval
from collections import Counter
from pathlib import Path
import pickle
import warnings

import geopandas as gpd
import pandas as pd

warnings.filterwarnings("ignore")

BASE_DIR = Path(".")
DATA_DIR = BASE_DIR / "data"
OUTPUT_DIR = BASE_DIR / "outputs"
PROJECT_CRS = "EPSG:4527"

COMMUNITY_PATH = DATA_DIR / "community_units" / "beijing_community.shp"
POI_DIR = DATA_DIR / "poi_shp"
CLUSTER_SHP_PATH = OUTPUT_DIR / "community_doc2vec_kmeans.shp"
CITY_POI_PATH = OUTPUT_DIR / "city_area_poi_english.pkl"
CLUSTER_POI_PATH = OUTPUT_DIR / "cluster_poi_type_seq3_mid_gat2_class_english.pkl"
EF_TABLE_PATH = OUTPUT_DIR / "EF_seq3_gat_2_class_english.xlsx"


def first_existing_column(frame: pd.DataFrame, candidates: list[str]) -> str:
    for col in candidates:
        if col in frame.columns:
            return col
    raise KeyError(f"None of the candidate columns exists: {candidates}")


def load_community() -> gpd.GeoDataFrame:
    community = gpd.read_file(COMMUNITY_PATH).to_crs(PROJECT_CRS)
    if "SQDM" not in community.columns:
        community = community.reset_index(drop=True)
        community["SQDM"] = range(len(community))
    community["areas"] = community.area
    return community[["SQDM", "areas", "geometry"]]


def load_all_poi() -> gpd.GeoDataFrame:
    frames = []
    for poi_path in sorted(POI_DIR.glob("*.shp")):
        poi = gpd.read_file(poi_path).to_crs(PROJECT_CRS)
        class_col = first_existing_column(poi, ["class", "mid_class", "class_mid", "中"])
        poi = poi[[class_col, "geometry"]].rename(columns={class_col: "class"})
        poi = poi[poi["class"].apply(lambda x: isinstance(x, str))]
        frames.append(poi)
    all_poi = pd.concat(frames, ignore_index=True)
    return gpd.GeoDataFrame(all_poi, geometry="geometry", crs=PROJECT_CRS)


def build_community_poi_table() -> pd.DataFrame:
    community = load_community()
    all_poi = load_all_poi()
    with (OUTPUT_DIR / "poi_all.pkl").open("wb") as f:
        pickle.dump(all_poi, f)

    city_area_poi = gpd.sjoin(community, all_poi, how="inner", predicate="intersects")
    city_area_poi = city_area_poi[["SQDM", "geometry", "areas", "class"]]
    with CITY_POI_PATH.open("wb") as f:
        pickle.dump(city_area_poi, f)
    return city_area_poi


def attach_poi_lists_to_clusters(city_area_poi: pd.DataFrame) -> pd.DataFrame:
    clusters = gpd.read_file(CLUSTER_SHP_PATH).to_crs(PROJECT_CRS)
    if "SQDM" not in clusters.columns and "ID" in clusters.columns:
        clusters = clusters.rename(columns={"ID": "SQDM"})
    clusters = clusters.rename(columns={"SQDM": "ID"})
    if "k_6" not in clusters.columns:
        raise KeyError("The clustering shapefile must contain a k_6 column.")

    poi_lists = city_area_poi.groupby("SQDM")["class"].apply(list).reset_index()
    poi_lists = poi_lists.rename(columns={"SQDM": "ID", "class": "class_list"})
    cluster_poi_type = clusters.merge(poi_lists, how="inner", on="ID")
    with CLUSTER_POI_PATH.open("wb") as f:
        pickle.dump(cluster_poi_type, f)
    return cluster_poi_type


def normalize_class_list(value) -> list[str]:
    if isinstance(value, list):
        return value
    if isinstance(value, str):
        try:
            parsed = literal_eval(value)
            return parsed if isinstance(parsed, list) else []
        except Exception:
            return []
    return []


def compute_ef(cluster_poi_type: pd.DataFrame) -> pd.DataFrame:
    records = []
    for cluster_id, group in cluster_poi_type.groupby("k_6"):
        all_poi_classes = []
        for poi_list in group["class_list"]:
            all_poi_classes.extend(normalize_class_list(poi_list))
        for poi_class, count in Counter(all_poi_classes).items():
            records.append({"zone_id": cluster_id, "poi_class": poi_class, "count": count})

    counts = pd.DataFrame(records)
    total_poi = counts["count"].sum()
    zone_totals = counts.groupby("zone_id")["count"].sum().to_dict()
    class_totals = counts.groupby("poi_class")["count"].sum().to_dict()

    def ef(row: pd.Series) -> float:
        return (row["count"] / zone_totals[row["zone_id"]]) / (class_totals[row["poi_class"]] / total_poi)

    counts["EF"] = counts.apply(ef, axis=1)
    return counts


def main() -> None:
    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    city_area_poi = build_community_poi_table()
    cluster_poi_type = attach_poi_lists_to_clusters(city_area_poi)
    ef_table = compute_ef(cluster_poi_type)
    ef_pivot = ef_table.pivot_table(index="poi_class", columns="zone_id", values="EF", fill_value=0)
    ef_pivot.to_excel(EF_TABLE_PATH, sheet_name="EF")


if __name__ == "__main__":
    main()
