# -*- coding: utf-8 -*-
"""
Clean subway smart-card OD records and export standardized subway trip records.

Input files:
  data/transport_raw/transit/beijing_transit.csv
  data/study_area_boundary/beijing_central_area.shp  (optional, for endpoint filtering)

Output file:
  outputs/transport/subway_trips.csv
"""

from pathlib import Path
import math
import warnings

import geopandas as gpd
import pandas as pd

warnings.filterwarnings("ignore")

BASE_DIR = Path(".")
TRANSIT_PATH = BASE_DIR / "data" / "transport_raw" / "transit" / "beijing_transit.csv"
BOUNDARY_PATH = BASE_DIR / "data" / "study_area_boundary" / "beijing_central_area.shp"
OUTPUT_DIR = BASE_DIR / "outputs" / "transport"
OUTPUT_PATH = OUTPUT_DIR / "subway_trips.csv"

SUBWAY_CODE = "DT"
INPUT_CRS = "EPSG:4326"


def haversine_distance_m(lat1, lon1, lat2, lon2):
    radius_m = 6378137.0
    phi1 = math.radians(lat1)
    phi2 = math.radians(lat2)
    dphi = math.radians(lat2 - lat1)
    dlambda = math.radians(lon2 - lon1)
    a = math.sin(dphi / 2) ** 2 + math.cos(phi1) * math.cos(phi2) * math.sin(dlambda / 2) ** 2
    return 2 * radius_m * math.asin(math.sqrt(a))


def parse_transit_time(series: pd.Series) -> pd.Series:
    values = pd.to_numeric(series, errors="coerce").astype("Int64").astype(str)
    values = values.replace("<NA>", "")
    return pd.to_datetime(values, format="%Y%m%d%H%M%S", errors="coerce")


def filter_by_boundary(df: pd.DataFrame, boundary_path: Path) -> pd.DataFrame:
    if not boundary_path.exists():
        return df

    boundary = gpd.read_file(boundary_path).to_crs(INPUT_CRS)
    xmin, ymin, xmax, ymax = boundary.total_bounds
    return df[
        df["f_lon"].between(xmin, xmax)
        & df["f_lat"].between(ymin, ymax)
        & df["t_lon"].between(xmin, xmax)
        & df["t_lat"].between(ymin, ymax)
    ].copy()


def load_transit_records(path: Path) -> pd.DataFrame:
    if not path.exists():
        raise FileNotFoundError(f"Transit OD file not found: {path}")
    df = pd.read_csv(path)
    df = filter_by_boundary(df, BOUNDARY_PATH)
    return df


def clean_subway_records(df: pd.DataFrame) -> pd.DataFrame:
    data = df.copy()
    data["Stime"] = parse_transit_time(data["f_tm"])
    if "t_tm" in data.columns:
        data["Etime"] = parse_transit_time(data["t_tm"])
    else:
        data["Etime"] = pd.NaT

    if "f_mode" in data.columns:
        data = data.rename(columns={"f_mode": "mode"})
    data = data[data["mode"] == SUBWAY_CODE].copy()

    data = data.drop_duplicates(subset=["card_id", "f_lon", "f_lat", "t_lon", "t_lat"], keep="first")
    if {"fst_name", "tst_name"}.issubset(data.columns):
        data = data[data["fst_name"] != data["tst_name"]]

    data["distance_m"] = data.apply(
        lambda row: haversine_distance_m(row["f_lat"], row["f_lon"], row["t_lat"], row["t_lon"]),
        axis=1,
    )
    data = data[data["distance_m"].between(1000, 40000)]

    card_counts = data["card_id"].value_counts()
    valid_cards = card_counts[card_counts <= 10].index
    data = data[data["card_id"].isin(valid_cards)].copy()

    data = data.rename(columns={"f_lon": "O_lon", "f_lat": "O_lat", "t_lon": "D_lon", "t_lat": "D_lat"})
    data["mode"] = "Subway"
    data["duration_min"] = (data["Etime"] - data["Stime"]).dt.total_seconds() / 60
    data = data.sort_values(["card_id", "Stime"]).reset_index(drop=True)
    data["orderid"] = data.groupby("card_id").cumcount() + 1

    optional_cols = [col for col in ["fst_name", "tst_name", "f_line", "t_line"] if col in data.columns]
    return data[
        [
            "card_id",
            "orderid",
            "mode",
            "Stime",
            "Etime",
            "O_lon",
            "O_lat",
            "D_lon",
            "D_lat",
            "distance_m",
            "duration_min",
        ]
        + optional_cols
    ]


def main() -> None:
    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    records = load_transit_records(TRANSIT_PATH)
    subway_trips = clean_subway_records(records)
    subway_trips.to_csv(OUTPUT_PATH, index=False, encoding="utf_8_sig")


if __name__ == "__main__":
    main()
